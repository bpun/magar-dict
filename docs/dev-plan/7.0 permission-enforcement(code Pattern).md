Every admin route passes through one dependency, so permissions are defined in one place and tested once.


# app/core/deps.py
from enum import StrEnum
from fastapi import Depends, HTTPException, status

class Role(StrEnum):
    USER = "user"
    ADMIN = "admin"

def get_optional_user(token: str | None = Depends(oauth2_optional)) -> User | None:
    """Returns None for guests. Never raises, so public routes stay public."""
    ...

def get_current_user(user: User | None = Depends(get_optional_user)) -> User:
    if user is None or not user.is_active or not user.email_verified:
        raise HTTPException(status.HTTP_401_UNAUTHORIZED, "Login required")
    return user

def require_role(*roles: Role):
    def checker(user: User = Depends(get_current_user)) -> User:
        if user.role not in roles:
            raise HTTPException(status.HTTP_403_FORBIDDEN, "Insufficient permissions")
        return user
    return checker

CurrentUser = Annotated[User, Depends(get_current_user)]
OptionalUser = Annotated[User | None, Depends(get_optional_user)]
AdminUser = Annotated[User, Depends(require_role(Role.ADMIN))]

# app/api/public/words.py
@router.get("/words/{slug}", response_model=WordOut)
def get_word(slug: str, user: OptionalUser, db: DbSession):
    query = select(Word).where(Word.slug == slug, Word.deleted_at.is_(None))
    if user is None or user.role != Role.ADMIN:
        query = query.where(Word.status == WordStatus.PUBLISHED)  # guests never see drafts
    word = db.scalar(query)
    if not word:
        raise HTTPException(404, "Word not found")
    return word
